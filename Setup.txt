function runFullSetup() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  setupEmployeesSheet(ss);
  setupRequestsSheet(ss);
  setupConfigSheet(ss);
  removeDefaultSheet(ss);

  SpreadsheetApp.getUi().alert(
    '✅ Setup complete!\n\n' +
    'Next steps:\n' +
    '1. Fill in real employee data in the Employees tab\n' +
    '2. Paste Code.gs into a new Apps Script file\n' +
    '3. Deploy as Web App\n' +
    '4. Set up your Form and Triggers'
  );
}

// ------------------------------------------------------------
function setupEmployeesSheet(ss) {
  let sheet = ss.getSheetByName('Employees');
  if (sheet) ss.deleteSheet(sheet);
  sheet = ss.insertSheet('Employees');

  // Headers
  const headers = ['Email', 'Manager Email', 'Leave Balance'];
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

  // Style headers
  sheet.getRange(1, 1, 1, headers.length)
    .setBackground('#4a86e8')
    .setFontColor('#ffffff')
    .setFontWeight('bold')
    .setFontSize(11);

  // Sample data rows
  const samples = [
    ['jane@company.com',  'manager@company.com', 22],
    ['john@company.com',  'manager@company.com', 22],
    ['mary@company.com',  'manager@company.com', 22],
  ];
  sheet.getRange(2, 1, samples.length, samples[0].length).setValues(samples);

  // Column widths
  sheet.setColumnWidth(1, 220);
  sheet.setColumnWidth(2, 220);
  sheet.setColumnWidth(3, 130);

  // Freeze header row
  sheet.setFrozenRows(1);

  // Balance column — center align
  sheet.getRange(2, 3, 100, 1).setHorizontalAlignment('center');

  // Alternate row shading
  sheet.getRange(2, 1, samples.length, headers.length)
    .setBackground('#f8f9fa');

  Logger.log('✅ Employees sheet created');
}

// ------------------------------------------------------------
function setupRequestsSheet(ss) {
  let sheet = ss.getSheetByName('Requests');
  if (sheet) ss.deleteSheet(sheet);
  sheet = ss.insertSheet('Requests');

  const headers = [
    'Request ID',
    'Employee Email',
    'Employee Name',
    'Start Date',
    'End Date',
    'Days',
    'Notes',
    'Status',
    'Manager Comment',
    'Submitted At',
    'Decided At'
  ];
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

  // Style headers
  sheet.getRange(1, 1, 1, headers.length)
    .setBackground('#4a86e8')
    .setFontColor('#ffffff')
    .setFontWeight('bold')
    .setFontSize(11);

  // Column widths
  const widths = [160, 220, 160, 110, 110, 60, 200, 100, 200, 160, 160];
  widths.forEach((w, i) => sheet.setColumnWidth(i + 1, w));

  // Status column — center align
  sheet.getRange(2, 8, 1000, 1).setHorizontalAlignment('center');

  // Days column — center align
  sheet.getRange(2, 6, 1000, 1).setHorizontalAlignment('center');

  // Freeze header row
  sheet.setFrozenRows(1);

  // Conditional formatting for Status column (col 8)
  const range      = sheet.getRange(2, 8, 1000, 1);
  const rules      = [];
  const approved   = SpreadsheetApp.newConditionalFormatRule()
    .whenTextEqualTo('APPROVED')
    .setBackground('#d9ead3')
    .setFontColor('#38761d')
    .setRanges([range])
    .build();
  const rejected   = SpreadsheetApp.newConditionalFormatRule()
    .whenTextEqualTo('REJECTED')
    .setBackground('#fce5cd')
    .setFontColor('#b45f06')
    .setRanges([range])
    .build();
  const pending    = SpreadsheetApp.newConditionalFormatRule()
    .whenTextEqualTo('PENDING')
    .setBackground('#fff2cc')
    .setFontColor('#7f6000')
    .setRanges([range])
    .build();

  rules.push(approved, rejected, pending);
  sheet.setConditionalFormatRules(rules);

  Logger.log('✅ Requests sheet created');
}

// ------------------------------------------------------------
function setupConfigSheet(ss) {
  let sheet = ss.getSheetByName('Config');
  if (sheet) ss.deleteSheet(sheet);
  sheet = ss.insertSheet('Config');

  const headers = ['Key', 'Value'];
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

  sheet.getRange(1, 1, 1, headers.length)
    .setBackground('#4a86e8')
    .setFontColor('#ffffff')
    .setFontWeight('bold')
    .setFontSize(11);

  const config = [
    ['Annual PTO Days', 22],
    ['System Version',  '1.0'],
    ['Last Reset',      ''],
  ];
  sheet.getRange(2, 1, config.length, 2).setValues(config);

  sheet.setColumnWidth(1, 180);
  sheet.setColumnWidth(2, 180);
  sheet.setFrozenRows(1);

  Logger.log('✅ Config sheet created');
}

// ------------------------------------------------------------
function removeDefaultSheet(ss) {
  const defaultSheet = ss.getSheetByName('Sheet1');
  if (defaultSheet) {
    // Only remove if other sheets exist
    if (ss.getSheets().length > 1) {
      ss.deleteSheet(defaultSheet);
    }
  }
}
