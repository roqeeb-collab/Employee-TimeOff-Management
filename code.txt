function getSheet(name) {
  return SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(name);
}

function getEmployeeByEmail(email) {
  const sheet = getSheet('Employees');
  const data  = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][COL.EMP_EMAIL - 1].toString().trim().toLowerCase() === email.trim().toLowerCase()) {
      return { row: i + 1, data: data[i] };
    }
  }
  return null;
}

function getRequestById(requestId) {
  const sheet = getSheet('Requests');
  const data  = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][COL.REQ_ID - 1].toString() === requestId.toString()) {
      return { row: i + 1, data: data[i] };
    }
  }
  return null;
}

function generateRequestId() {
  return 'REQ-' + new Date().getTime();
}

function formatDate(date) {
  if (!date) return '';
  return Utilities.formatDate(new Date(date), Session.getScriptTimeZone(), 'dd MMM yyyy');
}

// WORKING DAYS CALCULATOR
// Counts weekdays between start and end dates (inclusive)
// Excludes Nigerian public holidays
// ============================================================
function calculateWorkingDays(startDate, endDate) {
  // Nigerian public holidays — extend this list each year
  const publicHolidays = [
    // 2026
    '2026-01-01', // New Year's Day
    '2026-01-03', // New Year (observed, if 1st falls on weekend)
    '2026-03-31', // Eid-el-Fitr (estimated)
    '2026-04-01', // Eid-el-Fitr (estimated)
    '2026-04-03', // Good Friday
    '2026-04-06', // Easter Monday
    '2026-05-01', // Workers' Day
    '2026-05-29', // Democracy Day
    '2026-06-06', // Eid-el-Kabir (estimated)
    '2026-06-07', // Eid-el-Kabir (estimated)
    '2026-08-26', // Eid-el-Mawlid (estimated)
    '2026-10-01', // Independence Day
    '2026-12-25', // Christmas Day
    '2026-12-26', // Boxing Day
    // 2027
    '2027-01-01', // New Year's Day
    '2027-03-20', // Eid-el-Fitr (estimated)
    '2027-03-21', // Eid-el-Fitr (estimated)
    '2027-03-26', // Good Friday
    '2027-03-29', // Easter Monday
    '2027-05-01', // Workers' Day
    '2027-05-27', // Eid-el-Kabir (estimated)
    '2027-05-28', // Eid-el-Kabir (estimated)
    '2027-05-29', // Democracy Day
    '2027-08-16', // Eid-el-Mawlid (estimated)
    '2027-10-01', // Independence Day
    '2027-12-25', // Christmas Day
    '2027-12-27', // Boxing Day (observed)
  ];

  const start   = new Date(startDate);
  const end     = new Date(endDate);
  const current = new Date(start);

  start.setHours(0, 0, 0, 0);
  end.setHours(0, 0, 0, 0);
  current.setHours(0, 0, 0, 0);

  let count = 0;
  while (current <= end) {
    const dayOfWeek  = current.getDay();
    const dateString = Utilities.formatDate(current, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    const isWeekend  = dayOfWeek === 0 || dayOfWeek === 6;
    const isHoliday  = publicHolidays.includes(dateString);
    if (!isWeekend && !isHoliday) count++;
    current.setDate(current.getDate() + 1);
  }

  return count;
}

// ============================================================
// QA VALIDATION
// Returns { valid: true } or { valid: false, reason: '...' }
// ============================================================
function validateRequest(empEmail, empName, startDate, endDate, balance) {
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  const start = new Date(startDate);
  const end   = new Date(endDate);
  start.setHours(0, 0, 0, 0);
  end.setHours(0, 0, 0, 0);

  // 1. Start date must not be in the past
  if (start < today) {
    return {
      valid: false,
      reason: 'Your start date (' + formatDate(startDate) + ') is in the past. '
            + 'Leave requests must be submitted for future dates.'
    };
  }

  // 2. End date must not be before start date
  if (end < start) {
    return {
      valid: false,
      reason: 'Your end date (' + formatDate(endDate) + ') is before your start date '
            + '(' + formatDate(startDate) + '). Please check your dates and resubmit.'
    };
  }

  // 3. Range must not exceed 30 calendar days (single request cap)
  const calendarDays = Math.round((end - start) / (1000 * 60 * 60 * 24)) + 1;
  if (calendarDays > 30) {
    return {
      valid: false,
      reason: 'Your requested date range spans ' + calendarDays + ' calendar days. '
            + 'A single request cannot exceed 30 calendar days. '
            + 'Please split this into separate requests or contact HR.'
    };
  }

  // 4. Working days must be greater than zero
  const workingDays = calculateWorkingDays(startDate, endDate);
  if (workingDays <= 0) {
    return {
      valid: false,
      reason: 'Your selected dates (' + formatDate(startDate) + ' to ' + formatDate(endDate) + ') '
            + 'contain no working days. They may fall entirely on weekends or public holidays. '
            + 'Please choose different dates.'
    };
  }

  // 5. Employee must have sufficient balance
  if (workingDays > balance) {
    return {
      valid: false,
      reason: 'You have requested ' + workingDays + ' working day(s) but your current leave balance is only '
            + balance + ' day(s). Please reduce your request or contact HR.'
    };
  }

  return { valid: true, days: workingDays };
}

// 1. FORM SUBMIT TRIGGER
// ============================================================
function onFormSubmit(e) {
  try {
    const values = e.values;
    // values[0] = Timestamp
    // values[1] = Employee Email
    // values[2] = Employee Name
    // values[3] = Start Date
    // values[4] = End Date
    // values[5] = Notes  (Number of Days field has been removed from the form)
    const empEmail  = values[1].trim();
    const empName   = values[2].trim();
    const startDate = values[3];
    const endDate   = values[4];
    const notes     = values[5] || '';

    // Check employee exists first so we can address them by name in error emails
    const employee = getEmployeeByEmail(empEmail);
    if (!employee) {
      rejectSubmission(
        empEmail,
        empName,
        'Your email address (' + empEmail + ') was not found in the system. '
        + 'Please contact HR to ensure your email is registered.'
      );
      return;
    }

    const mgrEmail = employee.data[COL.EMP_MGR - 1];
    const balance  = employee.data[COL.EMP_BAL - 1];

    // Run all QA checks
    const validation = validateRequest(empEmail, empName, startDate, endDate, balance);
    if (!validation.valid) {
      rejectSubmission(empEmail, empName, validation.reason);
      return;
    }

    const days      = validation.days;
    const reqSheet  = getSheet('Requests');
    const requestId = generateRequestId();

    reqSheet.appendRow([
      requestId, empEmail, empName,
      startDate, endDate, days, notes,
      'PENDING', '', new Date(), ''
    ]);

    notifyManager(requestId, empEmail, empName, mgrEmail, startDate, endDate, days, notes, balance);

  } catch (err) {
    Logger.log('onFormSubmit error: ' + err);
  }
}

function doGet(e) {
  const action    = e.parameter.action;
  const requestId = e.parameter.requestId;

  if (!action || !requestId) return serve('<h3>Invalid request.</h3>');

  if (action === 'approve' || action === 'reject') return showDecisionForm(requestId, action);
  if (action === 'confirm') return processDecision(e);

  return serve('<h3>Unknown action.</h3>');
}

function processDecision(e) {
  const requestId = e.parameter.requestId;
  const decision  = e.parameter.decision;
  const comment   = e.parameter.comment || '';

  const req = getRequestById(requestId);
  if (!req) return serve('<h3>Request not found.</h3>');

  const currentStatus = req.data[COL.REQ_STATUS - 1];
  if (currentStatus !== 'PENDING') {
    return HtmlService.createHtmlOutput(`
<!DOCTYPE html><html><head><meta charset="UTF-8"></head>
<body style="font-family:Arial,sans-serif;max-width:500px;margin:60px auto;padding:20px;text-align:center">
  <div style="font-size:56px">⚠️</div>
  <h2 style="color:#f57c00">Already Processed</h2>
  <p>This request has already been <strong>${currentStatus.toLowerCase()}</strong>.</p>
  <p style="color:#888;font-size:13px">No changes were made.</p>
</body></html>`)
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }

  const reqSheet = getSheet('Requests');
  const empEmail = req.data[COL.REQ_EMAIL - 1];
  const empName  = req.data[COL.REQ_NAME  - 1];
  const days     = parseInt(req.data[COL.REQ_DAYS - 1], 10);
  const status   = decision === 'approve' ? 'APPROVED' : 'REJECTED';

  reqSheet.getRange(req.row, COL.REQ_COMMENT).setValue(comment);

  const isApproved = decision === 'approve';

  const color      = isApproved ? '#34a853' : '#ea4335';

</head>

// 5. ANNUAL RESET
// ============================================================
function resetAnnualLeaveBalances() {
  const sheet = getSheet('Employees');
  const data  = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    sheet.getRange(i + 1, COL.EMP_BAL).setValue(ANNUAL_PTO);
  }
  const config     = getSheet('Config');
  const configData = config.getDataRange().getValues();
  for (let i = 1; i < configData.length; i++) {
    if (configData[i][0] === 'Last Reset') {
      config.getRange(i + 1, 2).setValue(new Date());
    }
  }
  Logger.log('Balances reset to ' + ANNUAL_PTO + ' for ' + (data.length - 1) + ' employees.');
}
